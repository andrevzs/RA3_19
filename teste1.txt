(6.0 7.0 *)             *{ multiplicação real }*
(9.0 4.0 |)             *{ divisão real }*
*{ ── operações inteiras ── }*
(17 5 /)                *{ divisão inteira }*
(17 5 %)                *{ resto da divisão }*
(2 10 ^)                *{ potenciação }*
(5 2.5 *)               *{ int × real = real (promoção numérica) }*
*{ ── variáveis: definição e uso ── }*

(SALDO)                 *{ carrega SALDO }*

((SALDO 10.0 -) 2.0 *)
(((3.0 4.0 *) 2.0 +) 5.0 -)    *{ 3 níveis de aninhamento }*
*{ ── operadores relacionais (produzem bool) ── }*
(SALDO 50.0 >)          *{ SALDO > 50? }*
(SALDO 100.0 <=)        *{ SALDO <= 100? }*
(3.0 5.0 >)             *{ relacional entre dois literais reais → bool }*
(3.0 SALDO >)           *{ relacional: literal como primeiro operando }*
*{ ── decisão ── }*
(IF (SALDO 0.0 >) (1.0 POSITIVO) (0.0 POSITIVO))
*{ ── repetição: acumula enquanto CONT < 5 ── }*
(0.0 CONT)
(WHILE (CONT 5.0 <) ((CONT 1.0 +) CONT))
